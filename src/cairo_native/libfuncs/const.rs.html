44/// Select and call the correct libfunc builder function from the selector.
45pub fn build<'ctx, 'this>(
46    context: &'ctx Context,
47    registry: &ProgramRegistry<CoreType, CoreLibfunc>,
48    entry: &'this Block<'ctx>,
49    location: Location<'ctx>,
50    helper: &LibfuncHelper<'ctx, 'this>,
51    metadata: &mut MetadataStorage,
52    selector: &ConstConcreteLibfunc,
53) -> Result<()> {
54    match selector {
55        ConstConcreteLibfunc::AsBox(info) => {
56            build_const_as_box(context, registry, entry, location, helper, metadata, info)
57        }
58        ConstConcreteLibfunc::AsImmediate(info) => {
59            build_const_as_immediate(context, registry, entry, location, helper, metadata, info)
60        }
61    }
62}

64/// Generate MLIR operations for the `const_as_box` libfunc.
65///
66/// The box is a pointer to the constant's global. This is sound because boxes are arena
67/// allocated and never written through or freed.
68pub fn build_const_as_box<'ctx, 'this>(
69    context: &'ctx Context,
70    registry: &ProgramRegistry<CoreType, CoreLibfunc>,
71    entry: &'this Block<'ctx>,
72    location: Location<'ctx>,
73    helper: &LibfuncHelper<'ctx, 'this>,
74    metadata: &mut MetadataStorage,
75    info: &ConstAsBoxConcreteLibfunc,
76) -> Result<()> {
77    let ptr = build_const_global_ptr(
78        context,
79        registry,
80        entry,
81        location,
82        helper,
83        metadata,
84        &info.const_type,
85    )?;
86
87    helper.br(entry, 0, &[ptr], location)
88}

90/// Generate MLIR operations for the `const_as_immediate` libfunc.
91pub fn build_const_as_immediate<'ctx, 'this>(
92    context: &'ctx Context,
93    registry: &ProgramRegistry<CoreType, CoreLibfunc>,
94    entry: &'this Block<'ctx>,
95    location: Location<'ctx>,
96    helper: &LibfuncHelper<'ctx, 'this>,
97    metadata: &mut MetadataStorage,
98    info: &ConstAsImmediateConcreteLibfunc,
99) -> Result<()> {
100    let const_type = as_const_type(registry, &info.const_type)?;
101    let (inner_type, inner_layout) =
102        registry.build_type_with_layout(context, helper, metadata, &const_type.inner_ty)?;
103
104    // Integers (including felts) are emitted as immediates: a load from the global would only
105    // be folded by LLVM at `OptLevel::Less` or above.
106    if inner_type.is_integer() {
107        let bytes = build_const_type_bytes(registry, const_type)?;
108        let value = entry.const_int_from_type(
109            context,
110            location,
111            BigUint::from_bytes_le(&bytes),
112            inner_type,
113        )?;
114
115        return helper.br(entry, 0, &[value], location);
116    }
117
118    let ptr = build_const_global_ptr(
119        context,
120        registry,
121        entry,
122        location,
123        helper,
124        metadata,
125        &info.const_type,
126    )?;
127    let value = entry.append_op_result(llvm::load(
128        context,
129        ptr,
130        inner_type,
131        location,
132        LoadStoreOptions::new().align(Some(IntegerAttribute::new(
133            IntegerType::new(context, 64).into(),
134            inner_layout.align() as i64,
135        ))),
136    ))?;
137
138    helper.br(entry, 0, &[value], location)
139}

141/// Returns a pointer to the read-only global holding the constant's native memory
142/// representation, emitting the global the first time the const type is seen in the module.
143fn build_const_global_ptr<'ctx, 'this>(
144    context: &'ctx Context,
145    registry: &ProgramRegistry<CoreType, CoreLibfunc>,
146    entry: &'this Block<'ctx>,
147    location: Location<'ctx>,
148    helper: &LibfuncHelper<'ctx, 'this>,
149    metadata: &mut MetadataStorage,
150    const_type_id: &ConcreteTypeId,
151) -> Result<Value<'ctx, 'this>> {
152    let symbol = format!("cairo_native__const_{}", const_type_id.id);
153
154    if metadata
155        .get_or_insert_with(ConstGlobalsMeta::default)
156        .mark_emitted(const_type_id)
157    {
158        let const_type = as_const_type(registry, const_type_id)?;
159        let align = registry
160            .get_type(&const_type.inner_ty)?
161            .layout(registry)?
162            .align();
163        let bytes = build_const_type_bytes(registry, const_type)?;
164
165        let i8_ty = IntegerType::new(context, 8).into();
166        let value = DenseElementsAttribute::new(
167            RankedTensorType::new(&[bytes.len() as u64], i8_ty, None).into(),
168            &bytes
169                .iter()
170                .map(|byte| IntegerAttribute::new(i8_ty, *byte as i8 as i64).into())
171                .collect::<Vec<_>>(),
172        )?;
173
174        helper.module.body().append_operation(
175            ods::llvm::GlobalOperationBuilder::new(context, location)
176                .initializer(Region::new())
177                .global_type(TypeAttribute::new(llvm::r#type::array(
178                    i8_ty,
179                    bytes.len() as u32,
180                )))
181                .sym_name(StringAttribute::new(context, &symbol))
182                .linkage(
183                    Attribute::parse(context, "#llvm.linkage<private>")
184                        .ok_or(Error::ParseAttributeError)?,
185                )
186                .value(value.into())
187                .constant(Attribute::unit(context))
188                .alignment(IntegerAttribute::new(
189                    IntegerType::new(context, 64).into(),
190                    align as i64,
191                ))
192                .build()
193                .into(),
194        );
195    }
196
197    Ok(entry.append_op_result(
198        ods::llvm::mlir_addressof(
199            context,
200            llvm::r#type::pointer(context, 0),
201            FlatSymbolRefAttribute::new(context, &symbol),
202            location,
203        )
204        .into(),
205    )?)
206}

208/// Serialize a constant into the native memory representation of its inner type.
209///
210/// This must match both the layouts in `TypeBuilder::layout` and the encoding in `Value::to_ptr`.
211///
212/// `info.inner_data` should be one of the following:
213/// - A single value, if the inner type is a simple numeric type (e.g., `felt252`, `u32`, etc.).
214/// - A list of const types, if the inner type is a struct. The type of each const type must be
215///   the same as the corresponding struct member type.
216/// - A selector (a single value) followed by a const type, if the inner type is an enum. The
217///   type of the const type must be the same as the corresponding enum variant type.
218fn build_const_type_bytes(
219    registry: &ProgramRegistry<CoreType, CoreLibfunc>,
220    info: &ConstConcreteType,
221) -> Result<Vec<u8>> {
222    let layout = registry.get_type(&info.inner_ty)?.layout(registry)?;
223
224    let mut bytes = Vec::with_capacity(layout.size());
225    build_const_type_bytes_into(registry, info, &mut bytes)?;
226    Ok(bytes)
227}

229/// Appends the constant's bytes to `bytes`. Offsets are relative to the buffer length on entry.
230fn build_const_type_bytes_into(
231    registry: &ProgramRegistry<CoreType, CoreLibfunc>,
232    info: &ConstConcreteType,
233    bytes: &mut Vec<u8>,
234) -> Result<()> {
235    let inner_type = registry.get_type(&info.inner_ty)?;
236    let layout = inner_type.layout(registry)?;
237
238    let start = bytes.len();
239    match inner_type {
240        CoreTypeConcrete::Struct(_) => {
241            for field in &info.inner_data {
242                let GenericArg::Type(const_field_ty) = field else {
243                    return Err(Error::ConstDataMismatch);
244                };
245                let const_field_type = as_const_type(registry, const_field_ty)?;
246                let field_align = registry
247                    .get_type(&const_field_type.inner_ty)?
248                    .layout(registry)?
249                    .align();
250
251                bytes.resize(
252                    start + (bytes.len() - start).next_multiple_of(field_align),
253                    0,
254                );
255                build_const_type_bytes_into(registry, const_field_type, bytes)?;
256            }
257        }
258        CoreTypeConcrete::Enum(enum_info) => match &info.inner_data[..] {
259            [GenericArg::Value(variant_index), GenericArg::Type(payload_ty)] => {
260                let variant_index: usize = variant_index
261                    .try_into()
262                    .map_err(|_| Error::IntegerConversion)?;
263                let (_, tag_layout, variant_layouts) =
264                    get_layout_for_variants(registry, &enum_info.variants)?;
265                let payload_offset = tag_layout.extend(variant_layouts[variant_index])?.1;
266
267                bytes.extend(&variant_index.to_le_bytes()[..tag_layout.size()]);
268                bytes.resize(start + payload_offset, 0);
269                build_const_type_bytes_into(registry, as_const_type(registry, payload_ty)?, bytes)?;
270            }
271            _ => return Err(Error::ConstDataMismatch),
272        },
273        // NonZero is the same type as the inner type in native.
274        CoreTypeConcrete::NonZero(_) => match &info.inner_data[..] {
275            [GenericArg::Type(inner)] => {
276                build_const_type_bytes_into(registry, as_const_type(registry, inner)?, bytes)?;
277            }
278            _ => return Err(Error::ConstDataMismatch),
279        },
280        CoreTypeConcrete::BoundedInt(BoundedIntConcreteType { range, .. }) => {
281            match &info.inner_data[..] {
282                [GenericArg::Value(value)] => {
283                    bytes.extend(range.repr_encode(value).ok_or_else(|| {
284                        CompilerError::BoundedIntOutOfRange {
285                            value: Box::new(value.clone()),
286                            range: Box::new((range.lower.clone(), range.upper.clone())),
287                        }
288                    })?);
289                }
290                _ => return Err(Error::ConstDataMismatch),
291            }
292        }
293        CoreTypeConcrete::Felt252(_)
294        | CoreTypeConcrete::Starknet(
295            StarknetTypeConcrete::ClassHash(_) | StarknetTypeConcrete::ContractAddress(_),
296        ) => match &info.inner_data[..] {
297            [GenericArg::Value(value)] => bytes.extend(felt_to_unsigned(value).to_bytes_le()),
298            _ => return Err(Error::ConstDataMismatch),
299        },
300        CoreTypeConcrete::EcPoint(_) => match &info.inner_data[..] {
301            [GenericArg::Value(x), GenericArg::Value(y)] => {
302                bytes.extend(felt_to_unsigned(x).to_bytes_le());
303                bytes.resize(start + get_integer_layout(252).size(), 0);
304                bytes.extend(felt_to_unsigned(y).to_bytes_le());
305            }
306            _ => return Err(Error::ConstDataMismatch),
307        },
308        CoreTypeConcrete::Uint8(_)
309        | CoreTypeConcrete::Uint16(_)
310        | CoreTypeConcrete::Uint32(_)
311        | CoreTypeConcrete::Uint64(_)
312        | CoreTypeConcrete::Uint128(_)
313        | CoreTypeConcrete::Sint8(_)
314        | CoreTypeConcrete::Sint16(_)
315        | CoreTypeConcrete::Sint32(_)
316        | CoreTypeConcrete::Sint64(_)
317        | CoreTypeConcrete::Sint128(_)
318        | CoreTypeConcrete::Bytes31(_) => match &info.inner_data[..] {
319            // Two's complement over the full layout size.
320            [GenericArg::Value(value)] => {
321                let modulus = BigInt::one() << (8 * layout.size());
322                bytes.extend(value.rem_euclid(&modulus).to_bytes_le().1);
323            }
324            _ => return Err(Error::ConstDataMismatch),
325        },
326        _ => native_panic!("const for type {} not implemented", info.inner_ty),
327    };
328
329    bytes.resize(start + layout.size(), 0);
330    Ok(())
331}

333fn as_const_type<'a>(
334    registry: &'a ProgramRegistry<CoreType, CoreLibfunc>,
335    type_id: &ConcreteTypeId,
336) -> Result<&'a ConstConcreteType> {
337    match registry.get_type(type_id)? {
338        CoreTypeConcrete::Const(inner) => Ok(inner),
339        _ => native_panic!("matched an unexpected CoreTypeConcrete that is not a Const"),
340    }
341}

355    fn run_const_as_box_aot() {
356        // The constant's global must survive AOT object emission.
357        let program = load_cairo! {
358            use core::box::BoxTrait;
359
360            fn run_test() -> Box<felt252> {
361                BoxTrait::new(1234)
362            }
363        };
364
365        let native_context = NativeContext::new();
366        let module = native_context
367            .compile(&program.1, false, Some(Default::default()), None)
368            .unwrap();
369        let executor = AotNativeExecutor::from_native_module(module, OptLevel::Aggressive).unwrap();
370        let result = executor
371            .invoke_dynamic(&program.1.funcs[0].id, &[], Some(u64::MAX))
372            .unwrap();
373        assert_eq!(result.return_value, Value::Felt252(1234.into()));
374    }
