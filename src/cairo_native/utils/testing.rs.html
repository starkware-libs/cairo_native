125/// Compile a cairo program found at the given path to sierra.
126pub fn cairo_to_sierra(program: &Path) -> crate::error::Result<Arc<Program>> {
127    if program
128        .extension()
129        .map(|x| {
130            x.to_ascii_lowercase()
131                .to_string_lossy()
132                .eq_ignore_ascii_case("cairo")
133        })
134        .unwrap_or(false)
135    {
136        cairo_lang_compiler::compile_cairo_project_at_path(
137            program,
138            CompilerConfig {
139                replace_ids: true,
140                ..Default::default()
141            },
142            InliningStrategy::Default,
143        )
144        .map_err(|err| crate::error::Error::ProgramParser(err.to_string()))
145    } else {
146        let source = std::fs::read_to_string(program)?;
147        ProgramParser::new()
148            .parse(&source)
149            .map_err(|err| crate::error::Error::ProgramParser(err.to_string()))
150    }
151    .map(Arc::new)
152}

178pub(crate) fn compile_contract(program_str: &str, mut db: RootDatabase) -> (String, ContractClass) {
179    let mut program_file = tempfile::Builder::new()

243pub fn run_program(
244    program: &(String, Program),
245    entry_point: &str,
246    args: &[Value],
247) -> ExecutionResult {
248    let entry_point = format!("{0}::{0}::{1}", program.0, entry_point);
249    let program = &program.1;
250
251    let entry_point_id = &program
252        .funcs
253        .iter()
254        .find(|x| x.id.debug_name.as_deref() == Some(&entry_point))
255        .expect("Test program entry point not found.")
256        .id;
257
258    let context = NativeContext::new();
259
260    let module = context
261        .compile(program, false, Some(Default::default()), None)
262        .expect("Could not compile test program to MLIR.");
263
264    let executor = JitNativeExecutor::from_native_module(module, OptLevel::Less).unwrap();
265    executor
266        .invoke_dynamic_with_syscall_handler(
267            entry_point_id,
268            args,
269            Some(u64::MAX),
270            &mut StubSyscallHandler::default(),
271        )
272        .unwrap()
273}

286/// Serializes a message into a vector of felts, the same way that Cairo
287/// serializes byte arrays. Used for asserting panic message on tests.
288///
289/// https://github.com/starkware-libs/cairo/tree/v2.12.3/corelib/src/debug.cairo#L142
290pub fn panic_byte_array(message: &str) -> Vec<Felt> {
291    // Prepend byte array magic, used to identify serialized `ByteArray` variables.
292    // https://github.com/starkware-libs/cairo/tree/v2.12.3/corelib/src/byte_array.cairo#L64
293    let mut array = vec![Felt::from_hex_unchecked(
294        "0x46a6158a16a947e5916b2a2ca68501a45e93d7110e81aa2d6438b1c57c879a3",
295    )];
296
297    let chunk_iter = message.bytes().chunks(31);
298    let mut chunks = chunk_iter.into_iter().collect_vec();
299
300    // Take last word as its serialized differently.
301    let pending = chunks
302        .pop()
303        .map(|pendign| pendign.collect_vec())
304        .unwrap_or_default();
305
306    // Serialize length of the byte array.
307    array.push(chunks.len().into());
308
309    // Serialize each byte array element.
310    for chunk in chunks {
311        let chunk = chunk.collect_vec();
312        array.push(Felt::from_bytes_be_slice(&chunk));
313    }
314
315    // Serialize last word with its length.
316    array.extend_from_slice(&[Felt::from_bytes_be_slice(&pending), pending.len().into()]);
317
318    array
319}

386/// Checks that, for every type declared in `program`, the Rust-side
387/// [`TypeBuilder::layout`] agrees with the size and alignment LLVM assigns to
388/// the MLIR type emitted by [`TypeBuilder::build`]. Returns one message per
389/// mismatching type.
390///
391/// Every declared type is built and emitted as an external global in a scratch
392/// module, which is then lowered and translated to LLVM IR so the real target
393/// data layout answers the question.
394pub fn find_type_layout_mismatches(
395    native_context: &NativeContext,
396    program: &Program,
397) -> Result<Vec<String>> {
398    const GLOBAL_PREFIX: &str = "cairo_native__layout_check__";
399
400    initialize_llvm_targets();
401
402    let context = native_context.context();
403    let registry = ProgramRegistry::<CoreType, CoreLibfunc>::new(program)?;
404    let location = Location::unknown(context);
405    let mut module = Module::new(location);
406    let mut metadata = MetadataStorage::new();
407
408    let mut expected = Vec::with_capacity(program.type_declarations.len());
409    for declaration in &program.type_declarations {
410        let concrete_type = registry.get_type(&declaration.id)?;
411        if matches!(concrete_type, CoreTypeConcrete::Const(_)) {
412            // Const types only exist at compile time and have no MLIR type.
413            continue;
414        }
415
416        let ty = match registry.build_type(context, &module, &mut metadata, &declaration.id) {
417            Ok(ty) => ty,
418            // Circuit gates and anything that contains them are compile-time only.
419            Err(Error::SierraAssert(SierraAssertError::BadTypeInit(_))) => continue,
420            Err(e) => return Err(e),
421        };
422        let layout = concrete_type.layout(&registry)?.pad_to_align();
423
424        module.body().append_operation(
425            ods::llvm::mlir_global(
426                context,
427                Region::new(),
428                TypeAttribute::new(ty),
429                StringAttribute::new(context, &format!("{GLOBAL_PREFIX}{}", expected.len())),
430                Attribute::parse(context, "#llvm.linkage<external>")
431                    .to_native_assert_error("linkage attribute should parse")?,
432                location,
433            )
434            .into(),
435        );
436        expected.push((&declaration.long_id, layout));
437    }
438
439    run_pass_manager(context, &mut module)?;
440
441    let data_layout_rep = CString::new(get_data_layout_rep()?)
442        .to_native_assert_error("data layout string should not contain nul bytes")?;
443
444    let mut mismatches = Vec::new();
445    unsafe {
446        let llvm_context = LLVMContextCreate();
447        let llvm_module: LLVMModuleRef =
448            mlirTranslateModuleToLLVMIR(module.as_operation().to_raw(), llvm_context as *mut _)
449                as *mut _;
450        LLVMSetDataLayout(llvm_module, data_layout_rep.as_ptr());
451        let data_layout = LLVMGetModuleDataLayout(llvm_module);
452
453        for (index, (type_id, layout)) in expected.iter().enumerate() {
454            let name = CString::new(format!("{GLOBAL_PREFIX}{index}"))
455                .to_native_assert_error("global name should not contain nul bytes")?;
456            let global = LLVMGetNamedGlobal(llvm_module, name.as_ptr());
457            assert!(
458                !global.is_null(),
459                "global for {type_id} was dropped during lowering"
460            );
461            let ty = LLVMGlobalGetValueType(global);
462            let llvm_size = LLVMABISizeOfType(data_layout, ty) as usize;
463            let llvm_align = LLVMABIAlignmentOfType(data_layout, ty) as usize;
464            if llvm_size != layout.size() || llvm_align != layout.align() {
465                mismatches.push(format!(
466                    "{type_id}: layout() reports size {} align {}, but LLVM reports size \
467                     {llvm_size} align {llvm_align}",
468                    layout.size(),
469                    layout.align(),
470                ));
471            }
472        }
473
474        LLVMDisposeModule(llvm_module);
475        LLVMContextDispose(llvm_context);
476    }
477
478    Ok(mismatches)
